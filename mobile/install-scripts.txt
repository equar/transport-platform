npm run android -- --env aws        # emulator against deployed AWS backend
npm run ios -- --env aws            # simulator against deployed AWS backend
npm run android -- --env local      # emulator against local backend (default)
npm run ios -- --env local          # simulator against local backend (default)

# profile shortcuts
npm run android:dev                  # local backend + debug profile
npm run android:prod                 # AWS backend + release profile
npm run ios:dev                      # local backend + debug profile
npm run ios:prod                     # AWS backend + release profile

# physical devices need a LAN IP for local:
LOCAL_API_BASE_URL=http://YOUR_LAN_IP:8082/api npm run android:device -- --env local
LOCAL_API_BASE_URL=http://YOUR_LAN_IP:8082/api npm run ios:device -- --env local

# physical devices against AWS (no LAN IP needed):
npm run android:device -- --env aws
npm run ios:device -- --env aws
npm run android:device:prod
npm run ios:device:prod
# Detect connected physical iPhones/iPads without building or installing:
npm run ios:device -- --list-devices
# Physical iOS installation requires your Apple signing team:
DEVELOPMENT_TEAM=YOUR_TEAM_ID npm run ios:device:prod
# Optionally select one device explicitly:
# IOS_DEVICE_UDID=YOUR_DEVICE_UDID DEVELOPMENT_TEAM=YOUR_TEAM_ID npm run ios:device:prod

# From the repository root: install release builds on connected devices and booted simulators.

# DEVELOPMENT_TEAM=5YW3888543 ./localReleaseInstall.sh
